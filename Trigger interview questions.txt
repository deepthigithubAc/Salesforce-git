1. Auto-populate Account Phone on Contact

Scenario: When a Contact is created, copy the parent Account Phone to Contact Phone if Contact Phone is blank.

trigger ContactTrigger on Contact (before insert) {

    Set<Id> accountIds = new Set<Id>();

    for(Contact con : Trigger.new){
        if(con.AccountId != null && String.isBlank(con.Phone)){
            accountIds.add(con.AccountId);
        }
    }

    Map<Id, Account> accMap = new Map<Id, Account>(
        [SELECT Id, Phone FROM Account WHERE Id IN :accountIds]
    );

    for(Contact con : Trigger.new){
        if(con.AccountId != null &&
           String.isBlank(con.Phone) &&
           accMap.containsKey(con.AccountId)){

            con.Phone = accMap.get(con.AccountId).Phone;
        }
    }
}


2. Prevent Deletion of Closed Opportunities

Scenario: Users should not delete an Opportunity if Stage = Closed Won.

trigger OpportunityTrigger on Opportunity (before delete) {

    for(Opportunity opp : Trigger.old){
        if(opp.StageName == 'Closed Won'){
            opp.addError('Closed Won Opportunities cannot be deleted.');
        }
    }
}

3. Update Account when Opportunity is Closed Won

Scenario: When an Opportunity becomes Closed Won, update Account Customer Status = Customer.

trigger OpportunityTrigger on Opportunity(after update) {

    Set<Id> accountIds = new Set<Id>();

    for(Opportunity opp : Trigger.new){

        Opportunity oldOpp = Trigger.oldMap.get(opp.Id);

        if(opp.StageName == 'Closed Won'
           && oldOpp.StageName != 'Closed Won'
           && opp.AccountId != null){

            accountIds.add(opp.AccountId);
        }
    }

    List<Account> accList = [
        SELECT Id, Customer_Status__c
        FROM Account
        WHERE Id IN :accountIds
    ];

    for(Account acc : accList){
        acc.Customer_Status__c = 'Customer';
    }

    update accList;
}

4. Roll-up Count of Contacts on Account

Scenario: Store total contacts in Account.Contact_Count__c.

trigger ContactTrigger on Contact (
    after insert,
    after delete,
    after undelete
) {

    Set<Id> accountIds = new Set<Id>();

    if(Trigger.isInsert || Trigger.isUndelete){
        for(Contact con : Trigger.new){
            if(con.AccountId != null)
                accountIds.add(con.AccountId);
        }
    }

    if(Trigger.isDelete){
        for(Contact con : Trigger.old){
            if(con.AccountId != null)
                accountIds.add(con.AccountId);
        }
    }

    List<Account> accs = [
        SELECT Id,
               (SELECT Id FROM Contacts)
        FROM Account
        WHERE Id IN :accountIds
    ];

    for(Account acc : accs){
        acc.Contact_Count__c = acc.Contacts.size();
    }

    update accs;
}

5. Prevent Duplicate Contacts by Email

Scenario: Don't allow duplicate Contact Email.

trigger ContactTrigger on Contact(before insert){

    Set<String> emails = new Set<String>();

    for(Contact con : Trigger.new){
        if(con.Email != null){
            emails.add(con.Email.toLowerCase());
        }
    }

    Map<String, Contact> existingEmails = new Map<String, Contact>();

    for(Contact con : [
        SELECT Id, Email
        FROM Contact
        WHERE Email IN :emails
    ]){
        existingEmails.put(con.Email.toLowerCase(), con);
    }

    for(Contact con : Trigger.new){
        if(con.Email != null &&
           existingEmails.containsKey(con.Email.toLowerCase())){

            con.addError('Email already exists.');
        }
    }
}

6. Create Follow-up Task when Case is Closed

Scenario: When Case Status becomes Closed, create a Task.

trigger CaseTrigger on Case(after update){

    List<Task> tasks = new List<Task>();

    for(Case cs : Trigger.new){

        Case oldCs = Trigger.oldMap.get(cs.Id);

        if(cs.Status == 'Closed'
           && oldCs.Status != 'Closed'){

            tasks.add(new Task(
                Subject = 'Follow-up with Customer',
                WhatId = cs.Id,
                Status = 'Not Started',
                Priority = 'Normal'
            ));
        }
    }

    if(!tasks.isEmpty()){
        insert tasks;
    }
}

7. Calculate Total Opportunity Amount on Account

Scenario: Update Account Total_Revenue__c whenever Opportunities are inserted/updated/deleted.

trigger OpportunityTrigger on Opportunity(
    after insert,
    after update,
    after delete
){

    Set<Id> accountIds = new Set<Id>();

    if(Trigger.isDelete){
        for(Opportunity opp : Trigger.old){
            accountIds.add(opp.AccountId);
        }
    }else{
        for(Opportunity opp : Trigger.new){
            accountIds.add(opp.AccountId);
        }
    }

    List<Account> accountsToUpdate = new List<Account>();

    for(AggregateResult ar : [
        SELECT AccountId accId,
               SUM(Amount) total
        FROM Opportunity
        WHERE AccountId IN :accountIds
        GROUP BY AccountId
    ]){
        accountsToUpdate.add(
            new Account(
                Id = (Id)ar.get('accId'),
                Total_Revenue__c = (Decimal)ar.get('total')
            )
        );
    }

    update accountsToUpdate;
}

8. Send Email When Status Changes

Scenario :Send email when Case is Closed.

trigger CaseTrigger on Case(after update){

    List<Messaging.SingleEmailMessage> emails =
        new List<Messaging.SingleEmailMessage>();

    for(Case c : Trigger.new){

        Case oldCase = Trigger.oldMap.get(c.Id);

        if(c.Status == 'Closed' &&
           oldCase.Status != 'Closed'){

            Messaging.SingleEmailMessage mail =
                new Messaging.SingleEmailMessage();

            mail.setToAddresses(
                new String[]{'admin@test.com'}
            );

            mail.setSubject('Case Closed');

            mail.setPlainTextBody(
                'Case Number : ' + c.CaseNumber
            );

            emails.add(mail);
        }
    }

    if(!emails.isEmpty()){
        Messaging.sendEmail(emails);
    }
}
